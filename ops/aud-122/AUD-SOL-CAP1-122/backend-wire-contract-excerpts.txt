   782	   * Throw a structured 429 when the caller has exhausted their 24h user-turn
   783	   * budget. Counts `user` turns in the rolling window, plus the content-free
   784	   * count of user turns erased by `deleteSession` from a recently active
   785	   * session (so deleting a chat never resets the cap; conservative: those

   789	  async assertWithinRateLimit(caller: RomanCaller): Promise<void> {
   790	    if (caller.role === 'owner') return;
   791	    const cap = this.rateLimitCapFor(caller);
   792	    const since = new Date(Date.now() - ROMAN_RATE_LIMIT_WINDOW_MS);
   793	    const live = await this.prisma.romanMessage.count({
   794	      where: {
   795	        user_id: caller.id,
   796	        role: 'user',
   797	        created_at: { gte: since },
   798	      },
   799	    });
   800	    // Content-free counts kept by erased shells (deleteSession / the sweep).
   801	    // Summed in the database, so any number of deleted chats counts in full.
   802	    // Unerased pre-upgrade tombstones are excluded: their messages still
   803	    // exist and are already in `live`.
   804	    const erased = await this.prisma.romanSession.aggregate({
   805	      where: {
   806	        user_id: caller.id,
   807	        deleted_at: { not: null },
   808	        day_key: { startsWith: ROMAN_ERASED_DAY_KEY_PREFIX },
   809	        last_activity_at: { gte: since },
   810	        message_count: { gt: 0 },
   811	      },
   812	      _sum: { message_count: true },
   813	      _min: { last_activity_at: true },
   814	    });
   815	    const used = live + (erased._sum.message_count ?? 0);
   816	    if (used >= cap) {
   817	      // Retry-after = time until the oldest counted turn (or erased session's
   818	      // last activity) falls out of the window.
   819	      const oldest = await this.prisma.romanMessage.findFirst({
   820	        where: {
   821	          user_id: caller.id,
   822	          role: 'user',
   823	          created_at: { gte: since },
   824	        },
   825	        orderBy: { created_at: 'asc' },
   826	      });
   827	      const candidates = [
   828	        ...(oldest ? [oldest.created_at.getTime()] : []),
   829	        ...(erased._min.last_activity_at ? [erased._min.last_activity_at.getTime()] : []),
   830	      ];
   831	      const earliest = candidates.length > 0 ? Math.min(...candidates) : null;
   832	      const retryAfterSeconds =
   833	        earliest !== null
   834	          ? Math.max(
   835	              1,
   836	              Math.ceil((earliest + ROMAN_RATE_LIMIT_WINDOW_MS - Date.now()) / 1000),
   837	            )
   838	          : Math.ceil(ROMAN_RATE_LIMIT_WINDOW_MS / 1000);
   839	      throw new HttpException(
   840	        {
   841	          code: ROMAN_ERROR_RATE_LIMIT,
   842	          retryAfterSeconds,
   843	          message: romanRateLimitMessage(retryAfterSeconds, caller.role),
   844	        },
   845	        HttpStatus.TOO_MANY_REQUESTS,
   846	      );
